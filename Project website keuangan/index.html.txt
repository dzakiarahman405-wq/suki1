    <h2> Dompet Digital Saya</h2>
    <div class="card"><h4>Saldo: <span id="balance">Rp 0</span></h4></div>
    
    <div class="card">
        <input type="text" id="text" placeholder="Keterangan (Cth: Gaji)">
        <input type="number" id="amount" placeholder="Jumlah (Cth: 50000)">
        <select id="type"><option value="income">Pemasukan (+)</option><option value="expense">Pengeluaran (-)</option></select>
        <button onclick="addTransaction()">Tambah</button>
    </div>

    <h3>Riwayat</h3>
    <ul id="list"></ul>

    <script>
        let transactions = JSON.parse(localStorage.getItem('transactions')) || [];
        const balanceEl = document.getElementById('balance');
        const listEl = document.getElementById('list');

        function formatRupiah(angka) { return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(angka); }

        function addTransaction() {
            const text = document.getElementById('text').value;
            const amount = parseFloat(document.getElementById('amount').value);
            const type = document.getElementById('type').value;
            
            if(text === '' || isNaN(amount)) { alert('Isi data dengan benar!'); return; }
            
            const finalAmount = type === 'expense' ? -amount : amount;
            transactions.push({ id: Date.now(), text, amount: finalAmount });
            
            updateLocalStorage();
            init();
            document.getElementById('text').value = '';
            document.getElementById('amount').value = '';
        }

        function removeTransaction(id) {
            transactions = transactions.filter(t => t.id !== id);
            updateLocalStorage();
            init();
        }

        function updateValues() {
            const amounts = transactions.map(t => t.amount);
            const total = amounts.reduce((acc, item) => acc + item, 0);
            balanceEl.innerText = formatRupiah(total);
        }

        function init() {
            listEl.innerHTML = '';
            transactions.forEach(t => {
                const li = document.createElement('li');
                li.className = t.amount < 0 ? 'minus' : 'plus';
                li.innerHTML = `${t.text} <span>${formatRupiah(Math.abs(t.amount))}</span> <button onclick="removeTransaction(${t.id})" style="width:auto; background:red; padding:5px;">X</button>`;
                listEl.appendChild(li);
            });
            updateValues();
        }

        function updateLocalStorage() { localStorage.setItem('transactions', JSON.stringify(transactions)); }
        init();

        // Mendaftarkan Service Worker untuk Offline
        if ('serviceWorker' in navigator) {
            navigator.serviceWorker.register('./service-worker.js');
        }
    </script>
